# AZURE APP SERVICE STARTUP COMMANDS
# Copy and paste ONE of these commands into the Azure Portal Configuration > Startup Command

# Option 1: Minimal startup (RECOMMENDED for REST-only)
python -m gunicorn --bind=0.0.0.0:$PORT --timeout=600 config.wsgi:application

# Option 2: With migrations
python manage.py migrate --noinput; python -m gunicorn --bind=0.0.0.0:$PORT --timeout=600 config.wsgi:application

# Option 3: Full startup with static files
python manage.py collectstatic --noinput; python manage.py migrate --noinput; python -m gunicorn --bind=0.0.0.0:$PORT --workers=2 --timeout=600 config.wsgi:application

# Option 4: Use startup script (if the script exists)
bash startup_simple.sh
